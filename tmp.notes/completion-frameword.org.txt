#+begin_src emacs-lisp
(use-package orderless
  :custom
  ;; Global completion styles with =basic= as a dynamic fallback.
  (completion-styles '(orderless basic))
  ;; Disables default category overrides to prevent external style injection.
  (completion-category-defaults nil)
  ;; Emacs 31+: Enables substring-like behavior for partial-completion.
  (completion-pcm-leading-wildcard t)
  ;; Strict global matching styles, omitting flex and initialism for performance.
  (orderless-matching-styles '(orderless-regexp orderless-literal))
  :config
  ;; Consolidates all category routing into a single alist to prevent
  ;; Elpaca async load-order overwrites from fragmented =add-to-list= calls.
  (setq completion-category-overrides
        `((file (styles basic partial-completion orderless))
          (eglot (styles ,@completion-styles))
          (eglot-capf (styles ,@completion-styles))))
  ;; Explicit affix dictionary for predictable Vim-like dispatch routing.
  (setq orderless-affix-dispatch-alist
        `((?! . orderless-without-literal)
          (?= . orderless-literal)
          (?% . orderless-initialism)
          (?~ . orderless-flex)
          (?, . orderless-literal-prefix)
          (?& . ,(if (fboundp 'orderless-annotation)
                     'orderless-annotation
                   'orderless-literal))))
  ;; Handles Consult disambiguation suffixes and file extension matching.
  (defun ar/orderless-disambiguation-dispatch (word _index _total)
    "Handle disambiguation suffixes for Consult and file extensions."
    (let ((tofu-re (if (boundp 'consult--tofu-regexp)
                       (concat consult--tofu-regexp "*\\'")
                     "\\'")))
      (cond
       ;; Exact end-of-string match for Consult commands.
       ((string-suffix-p "$" word)
        `(orderless-regexp . ,(concat (substring word 0 -1) tofu-re)))
       ;; File extension matching in minibuffer and shell modes.
       ((and (or minibuffer-completing-file-name
                 (derived-mode-p 'eshell-mode))
             (string-prefix-p "." word))
        `(orderless-regexp . ,(concat "\\." (substring word 1) tofu-re))))))
  ;; Chains the native affix dispatcher with the custom disambiguation dispatcher.
  (setq orderless-style-dispatchers
        '(orderless-affix-dispatch
          ar/orderless-disambiguation-dispatch)))
#+end_src

(use-package vertico
  :custom
  ;; Reserves exactly 10 lines of vertical space for the dropdown.
  (vertico-count 10)
  ;; Disables dynamic resizing to prevent UI jumping.
  (vertico-resize nil)
  ;; Allows cycling from the bottom of the list back to the top.
  (vertico-cycle t)
  :bind (:map vertico-map
              ;; Ido-like directory navigation.
              ("RET" . vertico-directory-enter)
              ("DEL" . vertico-directory-delete-char)
              ("M-DEL" . vertico-directory-delete-word)
              ;; Group navigation via arrow keys.
              ("C-<down>" . vertico-next-group)
              ("C-<up>" . vertico-previous-group)
              ;; Avy-style minibuffer jumps.
              ("M-q" . vertico-quick-jump)
              ("M-Q" . vertico-quick-exit)
              ("M-i" . vertico-quick-insert))
  :hook (;; Cleans up shadowed directory paths.
         (rfn-eshadow-update-overlay . vertico-directory-tidy)
         ;; Saves minibuffer state for session recovery.
         (minibuffer-setup . vertico-repeat-save))
  :config
  ;; Prevents cursor wandering behind the read-only prompt text.
  (setq minibuffer-prompt-properties
        '(read-only t cursor-intangible t face minibuffer-prompt))
  (add-hook 'minibuffer-setup-hook #'cursor-intangible-mode)
  ;; Allows opening a minibuffer inside a minibuffer.
  (setq enable-recursive-minibuffers t)
  ;; Displays depth indicator for nested recursive minibuffers.
  (minibuffer-depth-indicate-mode 1)
  ;; Activates the core Vertico engine post-load.
  (vertico-mode 1)
  ;; Activates the multiform extension mode.
  (vertico-multiform-mode 1)
  ;; Context-aware UI rules per command.
  (setq vertico-multiform-commands
        '((consult-imenu buffer)
          (consult-outline buffer)
          (consult-line-multi buffer)
          (consult-yasnippet grid)))
  ;; Dynamically applies text properties to candidates based on category.
  (defvar +vertico-transform-functions nil)
  (cl-defmethod vertico--format-candidate :around
    (cand prefix suffix index start &context ((not +vertico-transform-functions) null))
    (dolist (fun (ensure-list +vertico-transform-functions))
      (setq cand (funcall fun cand)))
    (cl-call-next-method cand prefix suffix index start))
  ;; Highlights directories with a trailing slash.
  (defun +vertico-highlight-directory-fn (file)
    "Highlight FILE as a directory if it ends with a slash."
    (when (string-suffix-p "/" file)
      (add-face-text-property 0 (length file) 'dired-directory 'append file))
    file)
  ;; Highlights enabled modes in M-x.
  (defun +vertico-highlight-enabled-mode-fn (cmd)
    "Highlight CMD if the corresponding mode is enabled."
    (let ((sym (intern cmd)))
      (with-current-buffer (nth 1 (buffer-list))
        (if (or (eq sym major-mode)
                (and (memq sym minor-mode-list)
                     (boundp sym)
                     (symbol-value sym)))
            (add-face-text-property 0 (length cmd) 'font-lock-constant-face 'append cmd)))
      cmd))
  ;; Applies directory highlighting to the file category.
  (add-to-list 'vertico-multiform-categories
               '(file (+vertico-transform-functions . +vertico-highlight-directory-fn)))
  ;; Applies mode highlighting to M-x.
  (add-to-list 'vertico-multiform-commands
               '(execute-extended-command (+vertico-transform-functions . +vertico-highlight-enabled-mode-fn)))
  ;; Renders Embark's action menu as a searchable grid.
  (add-to-list 'vertico-multiform-categories '(embark-keybinding grid))
  ;; Remaps C-s to fuzzy-search minibuffer input history.
  (defvar ar/minibuffer-maps
    '(minibuffer-local-map
      minibuffer-local-ns-map
      minibuffer-local-completion-map
      minibuffer-local-must-match-map
      minibuffer-local-isearch-map
      read-expression-map)
    "List of minibuffer keymaps.")
  (with-eval-after-load 'consult
    (dolist (keymap ar/minibuffer-maps)
      (define-key (symbol-value keymap) (kbd "C-s") #'consult-history))))
#+end_src
