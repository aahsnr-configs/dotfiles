** DONE Delete Pair
Configures the native =delete-pair= engine and provides an O(1) DWIM wrapper for adjacent, string, and nested pair deletion, strictly confined to programming modes.
#+begin_src emacs-lisp
;; Emacs 28+: Visual blink delay before deletion.
(setq delete-pair-blink-delay 0.1)

;; Emacs 31+: Push mark before deletion for inner-text recovery.
(when (boundp 'delete-pair-push-mark)
  (setq delete-pair-push-mark t))

(defun ar/delete-pair-dwim ()
  "Intelligently delete the pair surrounding or preceding the cursor.
Strictly confined to `prog-mode' to prevent structural destruction in prose."
  (interactive)
  (unless (derived-mode-p 'prog-mode)
    (user-error "Delete Pair DWIM restricted to prog-mode; use `ds` in prose"))
  (let ((prev (char-before))
        (next (char-after))
        (ppss (syntax-ppss)))
    (cond
     ;; Delete preceding closing delimiter or quote.
     ((and prev (memq (char-syntax prev) '(?\) ?\" ?\|)))
      (delete-pair -1))
     ;; Delete following opening delimiter or quote.
     ((and next (memq (char-syntax next) '(?\( ?\" ?\|)))
      (delete-pair 1))
     ;; Inside a string: jump to start, handle prefixes, and delete.
     ((nth 3 ppss)
      (let ((start (nth 8 ppss)))
        (goto-char start)
        ;; Handle string prefixes (e.g., f"..." in Python).
        (unless (memq (char-after) '(?\" ?\|))
          (forward-char 1))
        (delete-pair 1)))
     ;; Fallback: jump out of current list and delete enclosing pair.
     (t
      (condition-case nil
          (progn
            (backward-up-list)
            (delete-pair 1))
        (scan-error (user-error "Not inside a deletable pair")))))))
#+end_src

** DONE Grep
Finds text across files and allows direct editing of the search results to update the original files.
#+begin_src emacs-lisp
(use-package grep
  :straight (:type built-in)
  :commands (grep
             lgrep
             rgrep
             zgrep
             zrgrep)
  :custom
  ;; Prevents automatic buffer saving before search execution.
  (grep-save-buffers nil)
  ;; Groups search results by filename with visual separators.
  (grep-use-headings t)
  :config
  ;; `grep-apply-setting` persists templates across `grep-compute-defaults` cycles.
  (when-let* ((rg (executable-find "rg")))
    (grep-apply-setting 'grep-command (format "%s -nH --null --no-heading --color=never -e " rg))
    ;; `<X>` and `<C>` are dropped; `rg` lacks `--exclude` and ANSI colors corrupt `compilation-mode`.
    (grep-apply-setting 'grep-template
                        (format "%s -nH --null --no-heading --color=never -e <R> <F>" rg))
    (grep-apply-setting 'grep-find-command (format "%s -nH --null --no-heading --color=never -e " rg))
    ;; Routes `<X>` to `find` to bypass `rg`'s missing `--exclude` flag, then `-exec`s into `rg`.
    (grep-apply-setting 'grep-find-template
                        (format "find <D> <X> -type f <F> -exec %s -nH --null --no-heading --color=never -e <R> {} +" rg))
    ;; Routes `xref-find-references` natively to ripgrep's regex engine.
    (setq xref-search-program 'ripgrep))
  ;; Forces Evil into normal state for read-only navigation.
  (evil-set-initial-state 'grep-mode 'normal)
  ;; Forces Evil into normal state for the writable edit mode.
  (evil-set-initial-state 'grep-edit-mode 'normal)
  ;; Injects Vim "write and quit" / "quit without saving" parity for grep mutations.
  (evil-define-key 'normal grep-edit-mode-map
    "ZZ" #'grep-edit-save
    "ZQ" #'grep-edit-abort))
#+end_src
